## 1. Overview

The Aladdin SDS (Silent Deploy Service) is the service on HOST to detect, configure and upgrade for Datalogic HHS and FRS devices. It also provides the communication interface with Datalogic Connect system.

## 3. Aladdin SDS Configuration

```toml
# ============================================================================
# Datalogic Aladdin SDS Configuration File
# ============================================================================
# File Location:
#   Windows: C:\ProgramData\Datalogic\AladdinSDS\config.toml
#   Linux:   /usr/lib/AladdinSDS/config.toml
#
# Format: TOML (Tom's Obvious, Minimal Language)
#   - Strings must be in quotes: "value"
#   - Numbers have no quotes: 1234
#   - Booleans: true or false (no quotes)
#   - Paths on Windows: Use double backslashes "C:\\Path\\To\\File"

# ----------------------------------------------------------------------------
# [edge] - Edge Gateway Configuration
# ----------------------------------------------------------------------------
# Settings for communicating with the Datalogic edge gateway.
[edge]

# host - Edge gateway hostname or IP address
#   Network address of the edge gateway for cloud connectivity.
#   Values:
#     - "localhost": Gateway running on same machine (development/testing)
#     - IP address: Remote gateway (e.g., "192.168.1.100")
#     - Hostname: DNS-resolvable name (e.g., "edge-gateway.local")
host = "localhost"

# ----------------------------------------------------------------------------
# [aladdinSDS] - AladdinSDS Agent Behavior Configuration
# ----------------------------------------------------------------------------
# Controls how the SDS agent behaves, including telemetry and logging.
[aladdinSDS]

# sendTelemetryMs - Telemetry reporting interval (milliseconds)
#   How often the agent sends device telemetry data to the edge gateway.
#   Default: 3600000 (1 hour)
#   Range: 60000 (1 min) to 86400000 (24 hours)
#   Examples:
#     - 300000  = 5 minutes
#     - 3600000 = 1 hour (default)
#     - 7200000 = 2 hours
sendTelemetryMs = 3600000

# sendConnectionMin - Host/device connection re-publish interval (minutes)
#   How often the agent re-publishes a full connection snapshot to the edge
#   gateway. The snapshot covers:
#     - the host connection status for this machine
#     - every currently connected device (recursively, including children)
#   Useful to keep DL Connect's view of ONLINE hosts/devices fresh even if a
#   previous status publish was missed (e.g. transient broker disconnect).
#   The snapshot is only sent when the MQTT client is connected; cycles
#   where the broker is unreachable are skipped (no retry storm).
#   Default: 5 (every 5 minutes)
#   Range: 1 to 1440 (24 hours)
#   Set to 0 to DISABLE the periodic snapshot (no scheduler started).
#   Note: this is independent of the one-shot snapshot already sent on
#   (re)connect and on edge-host change.
#   Examples:
#     - 0  = disabled
#     - 1  = every minute (debug / smoke tests)
#     - 5  = every 5 minutes (default)
#     - 60 = every hour
sendConnectionMin = 5

# logLevel - Application logging verbosity
#   Controls how much detail is written to log files (restart required).
#   Valid values: "TRACE", "DEBUG", "INFO", "WARN", "ERROR", "FATAL", "OFF"
#   Default: "INFO"
logLevel = "INFO"

# ----------------------------------------------------------------------------
# [autoConfiguration] - Auto-Configuration Feature – Standalone Mode
# ----------------------------------------------------------------------------
# Settings for the auto-configuration feature – Standalone Mode.
[autoConfiguration]

# scheduleUpdateTime - Time window during which scheduled Auto-Configuration are allowed to run (firmware updates or configuration deployment).
#   Range: "HH:MM - HH:MM"
#   Default: "00:00 - 01:00"
scheduleUpdateTime = "00:00 - 01:00"

# eventDrivenUpdateMode - Controls behavior when an event-driven update is detected.
#   Valid values:
#   - "IMMEDIATE": Updates are applied instantly upon detection (ignore scheduleUpdateTime).
#   - "DEFER": Automatically updates are deferred until the current time falls within the scheduleUpdateTime window.
#   - "DISABLED": disables Auto-Configuration feature.
#   Default: "DEFER"
eventDrivenUpdateMode = "DEFER"

# retryTimes - Maximum number of retry attempts for the SDS package.
#   If the package still fails after reaching the maximum retry count, it will be blocked.
#   Default: 3
retryTimes = 3

# retryDelay - Delay between retry attempts in seconds.
#   Default: 30
retryDelay = 30

# removePackageAfterExecute - Determines whether the package is removed
#   This setting takes effect after all valid devices have been successfully processed.
#   The actual behavior depends on the selected value, which controls whether the package is removed, retained, or governed by its own configuration.
#   Valid values:
#     - "REMOVE": Remove the package after all valid devices have been configured.
#     - "NOT_REMOVE": Keep the package after execution.
#     - "DEPEND_ON_PACKAGE": Follow the package configuration.
#   Default: DEPEND_ON_PACKAGE
removePackageAfterExecute = "DEPEND_ON_PACKAGE"

# ----------------------------------------------------------------------------
# [files] - File Transfer Service Configuration
# ----------------------------------------------------------------------------
# HTTP server for receiving firmware packages and file uploads.
[files]

# protocol - File transfer protocol
#   Protocol used for file uploads from devices or external systems (restart required).
#   Valid values: "http", "https"
#   Default: "http"
protocol = "http"

# port - File service listening port
#   TCP port for the file transfer HTTP server (restart required).
#   Default: 8090
#   Range: 1024-65535 (avoid privileged ports below 1024)
port = 8090

# downloadTimeoutMs - File download timeout (milliseconds)
#   Maximum time allowed to download a firmware package (restart required).
#   Default: 3600000 (1 hour)
#   Range: 60000 (1 min) to unlimited
downloadTimeoutMs = 3600000

# downloadPath - Package download directory
#   Where SDS packages are downloaded (restart required).
#   Default: "download/"
#   Important: Ensure this directory has write permissions
#   Windows recommended: "C:\\ProgramData\\Datalogic\\AladdinSDS\\download"
#   Linux recommended:   "/var/lib/AladdinSDS/download"
downloadPath = "download/"

# ----------------------------------------------------------------------------
# [mqtt] - MQTT Broker Connection Configuration
# ----------------------------------------------------------------------------
# Connection settings for the MQTT message broker used to communicate
# with the edge gateway and cloud services.
[mqtt]

# port - MQTT broker port
#   TCP port of the MQTT broker (restart required).
#   Default: 1883
port = 1883

# ----------------------------------------------------------------------------
# [http] - HTTP API Server Configuration
# ----------------------------------------------------------------------------
# REST API server for device management, configuration, and monitoring.
[http]

# port - API server listening port (restart required).
#   Default: 9000
#   Range: 1024-65535
port = 9000

# ----------------------------------------------------------------------------
# [otel] - OpenTelemetry Export Configuration
# ----------------------------------------------------------------------------
# Settings used by the internal OTEL span/log exporter.
[otel]

# endpoint - OTLP collector endpoint
#   URL of the OTLP gRPC collector endpoint (restart required).
#   Default: "http://localhost:4317"
endpoint = "http://localhost:4317"

# serviceName - OpenTelemetry service name
#   Logical service name exported as resource attribute service.name (restart required).
#   Default: "SDS"
serviceName = "SDS"

# ----------------------------------------------------------------------------
# [connectivity] - Device Connectivity Policy
# ----------------------------------------------------------------------------
# Controls which communication channels Aladdin SDS uses for each device family.
[connectivity]

# FRSConnectivity - Connectivity policy for Fixed Retail Scanners (FRS)
#   Determines which communication channels are used for FRS devices (restart required).
#   Valid values:
#     - "NO": Disable all communication with FRS devices
#     - "HOST": Use only the primary host interface (USB-COM/USB-OEM)

#     - "NO": Disable all communication with HHS devices

#   Default: "BOTH"
HHSConnectivity = "BOTH"

# releaseHostFRS - Release host and service ports for FRS after operations complete

#   Stopping barcode reception on FRS device due to port disconnection.

releaseHostFRS = false

scheduleEnabled = false

triggerOnStart = false

manifestUrl = "https://aladdin.datalogic.com/sds/aladdin-sds-manifest.json"
```

## 4. HTTP Interface

### 4.1 Config
Service configuration management endpoints.

- GET `/config` → Get application configuration
- PUT `/config` → Update application configuration
- POST `/config/restore` → Restore default configuration

### 4.2 Device
Device management and command action endpoints.

- GET `/devices` → List devices
- GET `/devices/{deviceId}` → Get device
- POST `/devices/{deviceId}/command/beep` → Send beep command
- POST `/devices/{deviceId}/command/reset` → Send host command
- POST `/devices/{deviceId}/command/firmware-upgrade` → Send service command
- POST `/devices/{deviceId}/command/disable-receive-scan-data` → Disable receiving scan data from scanner
- POST `/devices/{deviceId}/command/enable-receive-scan-data` → Enable receiving scan data from scanner
- POST `/devices/{deviceId}/command/firmware-upgrade` → Send firmware update command
- POST `/devices/{deviceId}/command/host-command/disable` → Send disable host commands
- POST `/devices/{deviceId}/command/host-command/enable` → Send enable host commands
- POST `/devices/{deviceId}/command/logs` → Get logs
- POST `/devices/{deviceId}/command/reset` → Send reset command
- POST `/devices/{deviceId}/command/scan-response` → Send response for devices with two way communication to the device
- POST `/devices/{deviceId}/command/scanner/disable` → Send disable scanner command
- POST `/devices/{deviceId}/command/scanner/enable` → Send enable scanner command
- GET `/devices/{deviceId}/config` → Get config
- PUT `/devices/{deviceId}/config` → Save Config
- POST `/devices/{deviceId}/execute-zip-package` → Execute zip package
- GET `/devices/{deviceId}/health` → Get health
- GET `/devices/{deviceId}/info` → Get info
- GET `/devices/{deviceId}/stats` → Get stats
- PUT `/devices/{deviceId}/upload/` → Upload Config File
- POST `/devices/command/firmware-upgrade` → Upgrade firmware

### 4.3 Host
Host management endpoints.

- GET `/host/host-id` → Get host id
- POST `/host/host-id` → Set host id
- GET `/host/info` → Get info

### 4.4 Packages
Local packages directory management endpoints.

- GET `/packages` → List all packages
- DELETE `/packages` → Delete all packages
- POST `/packages` → Download a package

### 4.5 Service
Service control endpoints.

- POST `/service/restart` → Restart the service
- POST `/service/stop` → Stop the service
